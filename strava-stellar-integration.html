    Ejemplo de Integración con JavaScript

    // Función para conectar y obtener actividades async function fetchStravaActivities() { try { // 1. Autenticar con Strava const auth = await stravaAuth.login(); // 2. Obtener lista de actividades const activities = await stravaClient.getActivities(); // 3. Procesar cada actividad let totalKm = 0; activities.forEach(activity => { if (activity.type === 'Ride') { const km = activity.distance / 1000; // convertir metros a km totalKm += km; } }); // 4. Calcular CO2 evitado const CO2_PER_KM = 0.120; // kg const co2Avoided = totalKm * CO2_PER_KM; // 5. Emitir tokens en Stellar const tokensMint = await stellarBridge.mintTokens( walletAddress, co2Avoided // 1:1 ratio kg:tokens ); return { totalKm, co2Avoided, tokensMinted: tokensMint }; } catch (error) { console.error('Error en integración Strava:', error); } } // Ejecutar sincronización cada 24 horas setInterval(fetchStravaActivities, 24 * 60 * 60 * 1000);

    📜 Smart Contracts en Stellar

    ⚠️ Nota: Stellar utiliza Stellar Smart Contracts (Soroban) para automatizar la emisión de tokens.

    Configuración del Token BIKE

    // Configuración del asset BIKE en Stellar { "asset_code": "BIKE", "asset_issuer": "GBRIDE...", // Dirección del emisor "amount": "1000000000", // Supply máximo "issuer_public_key": "GBRIDE...", "issuer_secret_key": "SBRIDE..." // Solo en servidor } // Parámetros del token const BIKE = { code: 'BIKE', issuer: new PublicKey('GBRIDE...'), supply: '1000000000', decimals: 2 };

    Función de Minting Automático

    // Smart Contract Soroban (Rust) #[contractimpl] pub struct BikeRewards; #[contractimpl] impl BikeRewards { /// Emite tokens BIKE basado en CO2 evitado pub fn mint_bike_tokens( env: Env, user_wallet: Address, co2_kg: i128, ) -> Result { // Validar entrada if co2_kg <= 0 { return Err(Error::InvalidAmount); } // 1 kg CO2 = 1 BIKE token (con 2 decimales) let amount_to_mint = co2_kg * 100; // Emitir tokens let token = env.invoke_contract::( &env.current_contract_address(), &symbol_short!("mint"), invoke_contract_args![env, user_wallet, amount_to_mint], )?; // Registrar evento env.events().publish( (symbol_short!("mint"), user_wallet, co2_kg), amount_to_mint, ); Ok(amount_to_mint) } }

    Transferencias y Canjes

    // Transferencia de tokens entre usuarios async function transferBikeTokens( fromWallet, toWallet, amountTokens ) { const transaction = new TransactionBuilder(sourceAccount, { fee: BASE_FEE, networkPassphrase: Networks.PUBLIC_NETWORK, }) .addOperation( Operation.payment({ destination: toWallet, asset: Asset.native(), amount: amountTokens.toString(), }) ) .setTimeout(30) .build(); transaction.sign(Keypair.fromSecret(fromWallet)); return await server.submitTransaction(transaction); }
